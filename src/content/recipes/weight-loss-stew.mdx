import Step from '../../components/Step.astro';
import Ref from '../../components/Ref.astro';

<ol class="steps">

<Step id="prep_aromatics" title="Prep the aromatics">
  Chop the <Ref of="aromatics" /> into big pieces — this is a chunky stew, so
  keep everything with some bite.
</Step>

<Step id="start_base_oil" title="Start the base" when="has(added_fat, 'olive_oil')">
  Heat a Dutch oven over medium with the <Ref of="added_fat" /> — all the added
  fat for the whole pot. Add the <Ref of="aromatics" /> and let them soften.
</Step>

<Step id="start_base_dry" title="Start the base" when="!has(added_fat, 'olive_oil')">
  Heat a dry Dutch oven over medium and water-sauté: add the
  <Ref of="aromatics" /> with a splash of water and let them soften, adding a
  little more water whenever the pot starts to catch.
</Step>

<Step id="add_lentils" title="Add the lentils">
  Rinse the <Ref of="lentils" /> and stir them in — they'll thicken the broth
  as they cook.
</Step>

<Step id="add_protein" title="Add the protein" when="has(protein, 'tofu')">
  Drain the <Ref of="protein" fill="tofu" /> — no need to press it — and break
  it into the pot in big chunks.
</Step>

<Step id="hold_chicken" title="Hold the chicken back" when="has(protein, 'chicken_breast')">
  Don't add the <Ref of="protein" fill="chicken_breast" /> yet — breast dries
  out over a full simmer. Cut it bite-size and stir it in at the 20-minute
  mark; it cooks through in the last 10 minutes.
</Step>

<Step id="season" title="Season">
  Add the <Ref of="seasonings" /> and let them cook a few minutes so the
  flavors open up.
</Step>

<Step id="soy_umami" title="The umami shortcut" when="has(seasonings, 'soy_sauce')">
  The <Ref of="seasonings" fill="soy_sauce" /> is doing the slow-cooker's job —
  a few minutes of simmering reads like hours of braising.
</Step>

<Step id="add_ginger" title="Brighten it" when="has(extras, 'ginger')">
  Stir in the <Ref of="extras" fill="ginger" /> — a small spoonful brightens
  the whole pot.
</Step>

<Step id="rinse_mushrooms" title="Prep the mushrooms" when="has(vegetables, 'mushrooms')">
  Give the <Ref of="vegetables" fill="mushrooms" /> a quick rinse — no need to
  scrub or peel.
</Step>

<Step id="add_vegetables" title="Add the vegetables">
  Add the <Ref of="potato" /> and the <Ref of="vegetables" />, with everything
  that needs a knife cut into large chunks — big pieces keep the stew chunky.
  Let it all cook down for a moment.
</Step>

<Step id="tomato_juice" title="Tomatoes, juice and all" when="has(vegetables, 'diced_tomatoes')">
  The <Ref of="vegetables" fill="diced_tomatoes" /> go in juice and all — their
  liquid is the start of the broth.
</Step>

<Step id="simmer" title="Simmer">
  Cover with just enough water to reach the top of the ingredients. Put the lid
  on, drop to low heat, and simmer 25–30 minutes until the <Ref of="lentils" />
  are soft and the broth has thickened.
</Step>

<Step id="blend_half" title="Blend for creaminess" when="blend">
  Ladle about a third of the pot into a blender (or use a stick blender right
  in the pot), purée, and stir it back in — a creamy broth with zero cream.
</Step>

<Step id="finish_greens" title="Finish with greens">
  Off the heat, stir in the <Ref of="greens" /> and let them wilt. Serve in big
  bowls — one serving is a huge, filling portion.
</Step>

</ol>

### Tips

- The formula matters more than the exact recipe: **high volume, high fiber,
  enough protein, low calorie density.** Swap the vegetables, the protein, or
  the spices and you'll still hit the same goal.
- Liquid-based meals like stews and soups feel fuller for longer at far fewer
  calories than a dry meal of the same size — your stomach senses volume, not
  just calories.
- Reheated the next day, the potatoes turn into resistant starch — great for
  satiety and a little better for you than fresh.
- Use low-sodium soy sauce so you control the final salt level yourself.

### 小贴士

- 公式比配方重要：**大体积、高纤维、足量蛋白质、低热量密度。**
  换蔬菜、换蛋白质、换香料，目标不变就还是这道菜。
- 炖菜和汤这类流食，比同体积的干食热量低得多却更顶饱 — 胃感知的是体积，
  不只是热量。
- 第二天加热再吃，土豆会转化成抗性淀粉，更扛饿也更健康一点。
- 用低钠酱油，最终咸度自己把控。
