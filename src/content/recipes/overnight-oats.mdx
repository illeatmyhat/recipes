import Step from '../../components/Step.astro';
import Ref from '../../components/Ref.astro';

<ol class="steps">

<Step id="soak" title="Night before">
  Combine the <Ref of="oats" /> and <Ref of="chia" /> in a jar or container.
  Pour in the <Ref of="liquid" />, add the <Ref of="vanilla" /> and
  <Ref of="salt" />, and stir well. Cover and refrigerate overnight — six
  hours at minimum.
</Step>

<Step id="morning" title="Morning">
  Scoop the set base into a big bowl — it should be thick and creamy. Too
  thick? Loosen it with a splash of water or <Ref of="liquid" />.
</Step>

<Step id="add_fruit" title="Add the fruit">
  Top with the <Ref of="fruits" />, fresh or thawed from frozen. The delicate
  ones — banana, kiwi, raspberries — always go in fresh in the morning:
  overnight they brown, turn bitter, or fall apart.
</Step>

<Step id="finish" title="Finish" when="count(toppings) > 0">
  Add the <Ref of="toppings" /> and eat right away — the contrast between
  the cold, creamy base and whatever you put on top is the point of the
  bowl.
</Step>

<Step id="granola_last" title="Granola goes on last" when="has(toppings, 'granola')">
  The <Ref of="toppings" fill="granola" /> goes on right before the first
  spoonful — it turns soggy within minutes of touching the wet base.
</Step>

<Step id="seeds_last" title="Seeds just before eating" when="has(toppings, 'pumpkin_seeds')">
  Scatter the <Ref of="toppings" fill="pumpkin_seeds" /> on just before
  eating — they soften almost immediately on contact with the base.
</Step>

</ol>

<div class="lang-en-US">

### Tips

- Stays fresh for 3 days in the fridge. It gets tangier each day as the kefir
  cultures stay active.
- Frozen berries are nutritionally equivalent to fresh, and often cheaper.
- Too thick in the morning? Add a splash of water or kefir.
- Meal prep: store the oat-and-kefir base and the fruit separately, and add
  the fruit each morning.
